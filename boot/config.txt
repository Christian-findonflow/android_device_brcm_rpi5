# Early splash screen (shows before kernel loads)
disable_splash=1
splash=splash.png

# Boot optimization
boot_delay=0
initial_turbo=60
force_turbo=1

# Kernel
kernel=Image

# Ramdisk
initramfs ramdisk.img followkernel

# Camera - disabled for faster boot (enable if needed)
#camera_auto_detect=1

# Display
disable_overscan=1

# Display panel
# NEO: the official 7" panel is mounted upside down so the display cable
# leaves the bottom of the case. invx,invy make the kernel invert both touch
# axes (Android does not rotate touch for an install orientation), and the
# neo-panel-rot180 overlay (dts/neo-panel-rot180.dts) sets rotation=180 on
# the panel node so SurfaceFlinger rotates the output. splash.png is
# pre-rotated. ro.surface_flinger.primary_display_orientation does nothing
# here: with an AIDL composer SurfaceFlinger only asks the composer.
dtoverlay=vc4-kms-dsi-7inch,invx,invy
dtoverlay=neo-panel-rot180
#dtoverlay=vc4-kms-dsi-ili9881-5inch
#dtoverlay=vc4-kms-dsi-ili9881-7inch
#dtoverlay=vc4-kms-dsi-generic
dtparam=drm_fb0_rp1_dsi1=on

# Graphics acceleration
#dtoverlay=vc4-kms-v3d
dtoverlay=cma,cma-320

# I2C
# I2C bus 1 = the Grove ports on the Seeed CAN-FD HAT v2: ISM330DHCX lean
# sensor (0x6a) + BMP280 barometer (0x77), driven by the vehicle HAL over
# /dev/i2c-1 (no kernel sensor drivers involved).
dtparam=i2c_arm=on
dtparam=i2c_arm_baudrate=400000

# I2S
#dtparam=i2s=on

# SPI
dtparam=spi=on

# PCIe / NVMe (Pi 5)
dtparam=pciex1=on
dtoverlay=pciex1-compat-pi5

# Seeed CAN-FD HAT v2
dtoverlay=seeed-can-fd-hat-v2

# USB
dtoverlay=dwc2,dr_mode=peripheral
[cm5]
dtoverlay=dwc2,dr_mode=otg
[all]
dtoverlay=vc4-kms-v3d-pi5
